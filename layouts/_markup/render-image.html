{{- /* Bilder im Text: nur Dateien aus Seitenordnern (Page Bundles) im Inhalts-Repository. Üblich ist der zentrale Bilderordner, so schreibt es Pages CMS: "/bilder/name.jpg". Außerdem erlaubt: "name.jpg" (eigener Ordner) und "/artikel//name.jpg". Gesucht wird in jeder Sprache, denn Bilder ohne Sprachkennung gehören in Hugo zur Standardsprache. Alles andere (fremde Server, Pfade ohne Bundle) bricht ab. Jedes Bild wird in feste Breiten und WebP umgerechnet; Originale werden nicht veröffentlicht (publishResources=false). */ -}} {{- $p := .PageInner -}} {{- $d := .Destination -}} {{- $own := printf "/%s" (replace $p.File.Dir "\\" "/") -}} {{- $img := false -}} {{- if findRE `^[a-z0-9-]+\.(jpg|png|webp)$` $d 1 -}} {{- $img = $p.Resources.Get $d -}} {{- else -}} {{- with findRESubmatch `^/((?:[a-z0-9-]+/)+)([a-z0-9-]+\.(?:jpg|png|webp))$` $d 1 -}} {{- $dir := index (index . 0) 1 -}} {{- $name := index (index . 0) 2 -}} {{- if eq (printf "/%s" $dir) $own -}} {{- $img = $p.Resources.Get $name -}} {{- else -}} {{- /* anderer Ordner: Bundle in irgendeiner Sprache suchen */ -}} {{- $path := printf "/%s" (strings.TrimSuffix "/" $dir) -}} {{- range hugo.Sites -}} {{- if not $img -}} {{- with .GetPage $path }}{{ with .Resources.Get $name }}{{ $img = . }}{{ end }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if or (not $img) (ne $img.ResourceType "image") -}} {{- errorf "%s: Bild %q nicht gefunden – Bilder gehören nach content/bilder/ (im Text: /bilder/name.jpg)" $p.File.Path $d -}} {{- else -}} {{- $opts := "webp q80 photo" -}} {{- if eq $img.MediaType.SubType "png" }}{{ $opts = "webp q85 drawing" }}{{ end -}} {{- $srcset := slice -}} {{- $fallback := false -}} {{- range $w := slice 480 800 1200 1600 -}} {{- if le $w $img.Width -}} {{- $r := $img.Process (printf "resize %dx %s" $w $opts) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- if or (not $fallback) (le $w 800) }}{{ $fallback = $r }}{{ end -}} {{- end -}} {{- end -}} {{- if not $fallback -}} {{- $fallback = $img.Process $opts -}} {{- $srcset = slice (printf "%s %dw" $fallback.RelPermalink $fallback.Width) -}} {{- end -}} {{- $alt := .PlainText | default $p.Title -}} {{- if .IsBlock }}{{ end -}} {{- end -}}
